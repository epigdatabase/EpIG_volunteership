2. Data Completeness

Code 
key_fields <- c("decimalLatitude", "decimalLongitude", "year", "locality", 
                "minimumElevationInMeters", "family", "genus", "scientificName")

# Add human-readable labels
field_labels <- c(
  "decimalLatitude" = "Latitude",
  "decimalLongitude" = "Longitude",
  "year" = "Year",
  "locality" = "Locality",
  "minimumElevationInMeters" = "Elevation",
  "family" = "Family",
  "genus" = "Genus",
  "scientificName" = "Scientific Name"
)

completeness <- tibble(
  Field = key_fields,
  Count = sapply(key_fields, function(col) sum(!is.na(db[[col]]))),
  Completeness_Percent = sapply(key_fields, function(col) {
    round(sum(!is.na(db[[col]])) / nrow(db) * 100, 1)
  })
) %>%
  mutate(Field_Label = field_labels[Field]) %>%
  arrange(desc(Completeness_Percent)) %>%
  select(Field_Label, Count, Completeness_Percent)

completeness %>%
  kable(
    #caption = "Table 2. Field completeness of occurrence records",
    align = c("l", "r", "r"),
    col.names = c("Field", "Records Present", "Completeness (%)")
  ) %>%
  kable_styling(
    bootstrap_options = c("striped", "hover", "condensed"),
    full_width = FALSE,
    position = "left"
  ) %>%
  row_spec(0, bold = TRUE, background = "#f0f0f0") %>%
  # Color-code percentages
  column_spec(3, 
    color = case_when(
      completeness$Completeness_Percent >= 90 ~ "#2e7d32",  # Green
      completeness$Completeness_Percent >= 70 ~ "#ed6c02",  # Orange
      completeness$Completeness_Percent >= 50 ~ "#ed6c02",  # Orange
      TRUE ~ "#d32f2f"  # Red
    ),
    bold = TRUE
  ) %>%
  # Add footer note
  footnote(
    general = "Fields are ordered from most to least complete.",
    general_title = "Note: ",
    footnote_as_chunk = TRUE
  )

quality_flags <- tibble(

    "Records with missing coordinates",
    "Records with missing year",

  ),

    sum(is.na(db$decimalLatitude) | is.na(db$decimalLongitude)),

### Temporal distribution summary

# Extract year from eventDate and create decade groupings
temporal_summary <- db %>%
  mutate(
    # Extract year from eventDate (first 4 characters)
    year_extracted = as.numeric(substr(as.character(eventDate), 1, 4)),
    # Create decade from extracted year
    Decade = floor(year_extracted / 10) * 10,
    # Create decade label as range
    Decade_Label = paste0(Decade, " - ", Decade + 10)
  ) %>%
  # Filter out records with missing or invalid years
  filter(!is.na(Decade) & !is.na(year_extracted)) %>%
  group_by(Decade_Label, Decade) %>%
  summarise(
    Records = n(),
    Species = n_distinct(scientificName, na.rm = TRUE),
    .groups = "drop"
  ) %>%
  arrange(Decade) %>%
  mutate(Records = format(Records, big.mark = ",")) %>%
  select(-Decade)  # Remove helper column after sorting
